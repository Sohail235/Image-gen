Code Examples

JavaScript 
const generateImage = async (prompt, style = 'realistic') => {
  const response = await fetch('https://api.silentgen.com/v1/generate-image', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
    },
    body: JSON.stringify({
      prompt: prompt,
      style: style,
      resolution: '512x512',
      api_key: 'your_api_key_here'
    })
  });
  
  const data = await response.json();
  return data.image_url;
};
Python 
import requests

def generate_image(prompt, style='realistic'):
    response = requests.post(
        'https://api.silentgen.com/v1/generate-image',
        json={
            'prompt': prompt,
            'style': style,
            'resolution': '512x512',
            'api_key': 'your_api_key_here'
        }
    )
    data = response.json()
    return data['image_url']
